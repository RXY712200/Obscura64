cmake_minimum_required(VERSION 3.16)
project(Obscura64PackageConsumer LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Obscura64 CONFIG REQUIRED)
add_executable(consumer_c consumer.c)
add_executable(consumer_cpp consumer.cpp)
target_link_libraries(consumer_c PRIVATE Obscura64::obscura64)
target_link_libraries(consumer_cpp PRIVATE Obscura64::obscura64)
enable_testing()
add_test(NAME consumer_c COMMAND consumer_c)
add_test(NAME consumer_cpp COMMAND consumer_cpp)
